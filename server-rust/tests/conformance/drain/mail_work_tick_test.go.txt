package mailwork

import (
	"context"
	"log/slog"
	"os"
	"strconv"
	"testing"
	"setjoin.com/shared/pgxutil"
 "setjoin.com/mailserver/internal/webhook"
)

func TestGoFreeMailWorkTick(t *testing.T) {
	ctx, cancel := context.WithCancel(context.Background()); defer cancel()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_DRAIN_DSN"), 4, "", "gofree-mbox4-audit-hmac")
	if err != nil { t.Fatal(err) }
	defer pool.Close()
	w := NewWorker(pgxutil.WorkerPool{Pool: pool}, slog.Default())
 if os.Getenv("GOFREE_MAIL_WEBHOOKS")=="empty" {dispatcher:=webhook.NewDispatcher(pool,slog.Default());dispatcher.StartRetryWorker(ctx);w=NewWorker(pgxutil.WorkerPool{Pool:pool},slog.Default(),dispatcher)}
	var settle func()
	if os.Getenv("GOFREE_MAIL_WEBHOOKS") == "deliver" {
		dispatcher := webhook.GoFreeDispatcher(ctx, pool)
		w = NewWorker(pgxutil.WorkerPool{Pool: pool}, slog.Default(), dispatcher)
		settle = func() { webhook.GoFreeSettle(dispatcher, cancel) }
	}
	n, _ := strconv.Atoi(os.Getenv("GOFREE_MAIL_TICKS")); if n < 1 { n = 1 }
	for i:=0;i<n;i++ { w.drain(ctx) }
	if settle != nil {
		settle()
	}
}
