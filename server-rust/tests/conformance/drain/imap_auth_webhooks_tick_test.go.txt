// Recording-only overlay (#457): frozen Go's real IMAP server, with its real webhook
// Dispatcher, plays the steps of one recorded corpus session (GOFREE_IMAP_SESSION, from
// IMAP_ORACLE_DIR/go-sessions.json; the responses are ignored). The auth.login and
// auth.login_failed events reach the fake receivers named by the seeded subscriptions.
package imap

import (
	"bufio"
	"context"
	"encoding/json"
	"io"
	"log/slog"
	"net"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"
	"setjoin.com/mailserver/internal/webhook"
)

func goFreeIMAPRead(c net.Conn, r *bufio.Reader, until string) {
	c.SetReadDeadline(time.Now().Add(4 * time.Second))
	switch until {
	case "eof":
		_, _ = io.ReadAll(r)
		return
	case "none":
		c.SetReadDeadline(time.Now().Add(250 * time.Millisecond))
		_, _ = io.ReadAll(r)
		return
	}
	for {
		line, err := r.ReadString('\n')
		if err != nil || strings.HasPrefix(line, until+" ") {
			return
		}
	}
}

func TestGoFreeIMAPAuthWebhooks(t *testing.T) {
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	raw, err := os.ReadFile(filepath.Join(os.Getenv("IMAP_ORACLE_DIR"), "go-sessions.json"))
	if err != nil {
		t.Fatal(err)
	}
	var sessions []map[string]any
	if err = json.Unmarshal(raw, &sessions); err != nil {
		t.Fatal(err)
	}
	appPool, err := pgxpool.New(ctx, os.Getenv("DB_APP_DSN"))
	if err != nil {
		t.Fatal(err)
	}
	defer appPool.Close()
	// The server's own request pool also serves its dispatcher, as in the daemon.
	d := webhook.GoFreeDispatcher(ctx, appPool)
	for _, session := range sessions {
		if session["name"] != os.Getenv("GOFREE_IMAP_SESSION") {
			continue
		}
		l, err := net.Listen("tcp", "127.0.0.1:0")
		if err != nil {
			t.Fatal(err)
		}
		addr := l.Addr().String()
		l.Close()
		srvCtx, stop := context.WithCancel(ctx)
		s := NewServer(appPool, addr, nil, slog.New(slog.NewTextHandler(io.Discard, nil)), d, nil)
		if err = s.Start(srvCtx); err != nil {
			t.Fatal(err)
		}
		conns := map[int]net.Conn{}
		readers := map[int]*bufio.Reader{}
		for _, stepAny := range session["steps"].([]any) {
			step := stepAny.(map[string]any)
			if ms, ok := step["sleep_ms"].(float64); ok {
				time.Sleep(time.Duration(ms) * time.Millisecond)
			}
			idx := 0
			if v, ok := step["conn"].(float64); ok {
				idx = int(v)
			}
			if conns[idx] == nil {
				var c net.Conn
				for until := time.Now().Add(4 * time.Second); time.Now().Before(until); time.Sleep(10 * time.Millisecond) {
					if c, err = net.DialTimeout("tcp", addr, time.Second); err == nil {
						break
					}
				}
				if err != nil {
					t.Fatal(err)
				}
				conns[idx], readers[idx] = c, bufio.NewReader(c)
			}
			if req := step["request"].(string); req != "" {
				conns[idx].SetWriteDeadline(time.Now().Add(4 * time.Second))
				_, _ = conns[idx].Write([]byte(req))
			}
			goFreeIMAPRead(conns[idx], readers[idx], step["until"].(string))
		}
		for _, c := range conns {
			c.Close()
		}
		time.Sleep(100 * time.Millisecond)
		stop()
	}
	webhook.GoFreeSettle(d, cancel)
}
