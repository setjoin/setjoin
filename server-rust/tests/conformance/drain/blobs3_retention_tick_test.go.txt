// Recording-only overlay: frozen Go's actual PUT produces the retry entry;
// actual cleanup steps retry it. No shipped Go source is changed.
package blob

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"log/slog"
	"net/http"
	"net/http/httptest"
	"os"
	"strconv"
	"testing"

	"github.com/minio/minio-go/v7"
	"github.com/minio/minio-go/v7/pkg/credentials"
	"setjoin.com/mailserver/internal/auth"
	"setjoin.com/mailserver/internal/storage"
	"setjoin.com/shared/pgxutil"
)

func TestGoFreeBlobs3RetentionTicks(t *testing.T) {
	ctx := context.Background()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_DRAIN_DSN"), 4, "", "")
	if err != nil {
		t.Fatal(err)
	}
	defer pool.Close()
	endpoint := os.Getenv("GOFREE_DRAIN_S3")
	mc, err := minio.New(endpoint, &minio.Options{Creds: credentials.NewStaticV4("test-access", "test-secret", ""), Secure: false})
	if err != nil {
		t.Fatal(err)
	}
	secure := false
	driver, err := storage.NewDriver(storage.DriverConfig{Endpoint: "http://" + endpoint, Bucket: "mailblobs", AccessKey: "test-access", SecretKey: "test-secret", Secure: &secure, ObjectLockMode: "governance", ObjectLockRetentionDays: 30})
	if err != nil {
		t.Fatal(err)
	}
	retention, ok := storage.RetentionOf(driver)
	if !ok {
		t.Fatal("S3 retention capability missing")
	}
	h := NewHandler(mc, pool, pgxutil.WorkerPool{Pool: pool}, "mailblobs", slog.Default())
	h.SetRetentionRegistry(storage.NewRetentionPolicyRegistry(storage.NewWORMRetentionPolicy(retention, "governance", 30, nil)))
	user, err := strconv.ParseInt(os.Getenv("GOFREE_DRAIN_USER"), 10, 64)
	if err != nil || user <= 0 {
		t.Fatal("invalid fixture actor")
	}
	payload := []byte("blobs3 retry payload")
	digest := sha256.Sum256(payload)
	hash := hex.EncodeToString(digest[:])
	req := httptest.NewRequest(http.MethodPut, "/api/blobs/"+hash, bytes.NewReader(payload))
	req.SetPathValue("sha256", hash)
	req.Header.Set("Content-Type", "text/plain")
	req = req.WithContext(auth.ContextWithUser(ctx, &auth.Claims{UserID: user}))
	response := httptest.NewRecorder()
	h.Put(response, req)
	if response.Code != 201 {
		t.Fatalf("PUT status=%d body=%q", response.Code, response.Body.String())
	}
	observed, err := json.Marshal(map[string]any{"status": response.Code, "body": response.Body.String()})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("GOFREE_DRAIN_OUTPUT"), observed, 0600); err != nil {
		t.Fatal(err)
	}
	// First tick still refuses retention, second succeeds, subsequent ticks
	// must make no more retention requests. Observe only external S3 and rows.
	for i := 0; i < 4; i++ {
		h.processCleanupQueue(ctx)
		h.processOrphanedBlobs(ctx)
		h.retryFailedReaps(ctx)
		h.retryFailedRetention(ctx)
	}
}
