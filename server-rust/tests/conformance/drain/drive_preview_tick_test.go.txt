// Recording-only overlay (#409): runs frozen Go's real drive.PreviewWorker over
// the assigned database with a MinIO client on an owned fake S3 and the fake
// DRIVE_PREVIEW_CMD from the environment. Not a tracked Go change.
package drive

import (
	"context"
	"log/slog"
	"os"
	"strconv"
	"testing"
	"time"

	"github.com/minio/minio-go/v7"
	"github.com/minio/minio-go/v7/pkg/credentials"
	"setjoin.com/shared/pgxutil"
)

func TestGoFreeDrivePreviewTick(t *testing.T) {
	ctx := context.Background()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_DRAIN_DSN"), 4, "", "")
	if err != nil {
		t.Fatal(err)
	}
	defer pool.Close()
	mc, err := minio.New(os.Getenv("GOFREE_DRAIN_S3"), &minio.Options{
		Creds: credentials.NewStaticV4("test-access", "test-secret", ""), Secure: false,
	})
	if err != nil {
		t.Fatal(err)
	}
	if v, err := strconv.ParseInt(os.Getenv("GOFREE_PREVIEW_SRC_MAX"), 10, 64); err == nil {
		maxPreviewSourceBytes = v
	}
	if v, err := strconv.ParseInt(os.Getenv("GOFREE_PREVIEW_OUT_MAX"), 10, 64); err == nil {
		maxPreviewOutputBytes = v
	}
	w := NewPreviewWorker(pgxutil.WorkerPool{Pool: pool}, MinioStore{Client: mc}, "mailblobs",
		os.Getenv("DRIVE_PREVIEW_CMD"), slog.Default())
	if os.Getenv("GOFREE_PREVIEW_RUN") == "1" {
		// Run must return at once when the command is unset (inert).
		runCtx, cancel := context.WithTimeout(ctx, 20*time.Second)
		defer cancel()
		done := make(chan struct{})
		go func() { w.Run(runCtx); close(done) }()
		select {
		case <-done:
			if runCtx.Err() != nil {
				t.Fatal("Run did not return before the context ended")
			}
		case <-time.After(5 * time.Second):
			t.Fatal("Run did not return promptly")
		}
		return
	}
	w.tick(ctx)
}
