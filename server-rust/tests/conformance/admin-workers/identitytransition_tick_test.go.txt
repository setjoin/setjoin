// Recording-only overlay (#412): one iteration of frozen Go's real identity-transitions
// worker (effect drain, then session-eviction drain) against a real LiveKit client
// pointed at the oracle's fake SFU.
package identitytransition

import (
	"context"
	"log/slog"
	"os"
	"testing"

	"setjoin.com/mailserver/internal/meet"
	"setjoin.com/shared/pgxutil"
)

func TestGoFreeAdminJobsTick(t *testing.T) {
	ctx := context.Background()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_JOBS_DSN"), 4, "", os.Getenv("GOFREE_JOBS_HMAC"))
	if err != nil {
		t.Fatal(err)
	}
	defer pool.Close()
	var rooms meet.RoomController
	if url := os.Getenv("GOFREE_JOBS_LK_URL"); url != "" {
		rooms = meet.NewLKRooms(meet.Config{APIKey: "oracle-key", APISecret: "oracle-secret-oracle-secret-oracle", WSURL: url})
	}
	w := NewWorker(pgxutil.WorkerPool{Pool: pool}, rooms, slog.Default())
	w.drain(ctx)
	w.drainSessionEvictions(ctx)
}
