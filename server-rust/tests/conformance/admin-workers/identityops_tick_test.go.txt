// Recording-only overlay (#412): one tick of frozen Go's real identity-operations worker.
package identityops

import (
	"context"
	"log/slog"
	"os"
	"testing"

	"setjoin.com/shared/pgxutil"
)

func TestGoFreeAdminJobsTick(t *testing.T) {
	ctx := context.Background()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_JOBS_DSN"), 4, "", os.Getenv("GOFREE_JOBS_HMAC"))
	if err != nil {
		t.Fatal(err)
	}
	defer pool.Close()
	w := NewWorker(pgxutil.WorkerPool{Pool: pool}, slog.Default())
	if err := w.tick(ctx); err != nil {
		t.Logf("tick: %v", err)
	}
}
