// Recording-only overlay (#412): one tick of frozen Go's real email-address-policy-jobs
// worker (the startup reclaim of Run, then its drain loop).
package eapjobs

import (
	"context"
	"log/slog"
	"os"
	"testing"

	"setjoin.com/shared/pgxutil"
)

func TestGoFreeAdminJobsTick(t *testing.T) {
	ctx := context.Background()
	pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_JOBS_DSN"), 4, "", os.Getenv("GOFREE_JOBS_HMAC"))
	if err != nil {
		t.Fatal(err)
	}
	defer pool.Close()
	w := NewWorker(pgxutil.WorkerPool{Pool: pool}, slog.Default())
	if _, err := w.pool.Exec(ctx, `UPDATE eap_apply_jobs SET status='queued', updated_at=now() WHERE status='running'`); err != nil {
		t.Fatal(err)
	}
	for {
		claimed, err := w.processOne(ctx)
		if err != nil {
			t.Logf("processOne: %v", err)
			break
		}
		if !claimed {
			break
		}
	}
}
