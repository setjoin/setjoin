 enc:=base64.RawURLEncoding.EncodeToString

   }else if format=="android-safetynet"{

  params:=[]protocol.CredentialParameter{{Type:protocol.PublicKeyCredentialType,Algorithm:webauthncose.COSEAlgorithmIdentifier(alg)}}

   assertion,_:=json.Marshal(map[string]any{"id":enc(id),"rawId":enc(id),"type":"public-key","response":map[string]any{"clientDataJSON":enc(client),"authenticatorData":enc(data),"signature":enc(sig),"userHandle":enc(handle)}})

   label:="success";if count==6{label="regression"};if count==0{label="zero"}

  }
  out=append(out,map[string]any{"algorithm":alg,"id":enc(id),"key":hex.EncodeToString(key),"handle":enc(handle),"challenge":challenge,"registration":json.RawMessage(registration),"attestation_type":cred.AttestationType,"assertions":assertions})
 }
