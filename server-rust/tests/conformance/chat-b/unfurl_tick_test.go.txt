// Recording-only overlay: invokes frozen Go's actual private worker tick.
package chat

import (
 "context"
 "log/slog"
 "os"
 "testing"
 "setjoin.com/shared/pgxutil"
)

func TestGoFreeChat3ReapTick(t *testing.T) {
 ctx := context.Background()
 pool, err := pgxutil.OpenPool(ctx, os.Getenv("GOFREE_CHAT_WORKER_DSN"), 1, "", "")
 if err != nil { t.Fatal(err) }
 defer pool.Close()
 NewWorker(pgxutil.WorkerPool{Pool:pool}, slog.Default()).evictExpired(ctx)
}
