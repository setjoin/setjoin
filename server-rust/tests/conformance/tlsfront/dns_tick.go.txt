type goFreeDNSTransport struct {next http.RoundTripper; endpoint *url.URL}
func (tr goFreeDNSTransport)RoundTrip(r *http.Request)(*http.Response,error){
 if r.URL.Host=="api.cloudflare.com" {copy:=r.Clone(r.Context());u:=*r.URL;copy.URL=&u;copy.URL.Scheme=tr.endpoint.Scheme;copy.URL.Host=tr.endpoint.Host;return tr.next.RoundTrip(copy)}
 return tr.next.RoundTrip(r)
}
