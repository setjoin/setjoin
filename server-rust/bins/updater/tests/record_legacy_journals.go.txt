// Test-only recorder for the installer boundary. Production Go stays unchanged.
package main
import (
 "encoding/json"
 "os"
 "path/filepath"
 "fmt"
 "setjoin.com/mailserver/internal/updater"
)
func init() {
 if os.Getenv("REL3_RECORD_JOURNALS") == "" { return }
 cases := map[string]updater.State{"active":updater.StateMaintenance,"verified":updater.StateVerified,"recovery-required":updater.StateRecoveryRequired}
 result := map[string]interface{}{}
 for name,state := range cases {
  dir,err:=os.MkdirTemp("", "gofree-rel3-legacy-"); if err!=nil {panic(err)}
  journal,err:=updater.OpenJournal(dir); if err!=nil {panic(err)}
  _,err=journal.Append(updater.Event{IntentID:"42",ReleaseID:"fixture",ManifestDigest:"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa",LeaseID:"11111111-1111-4111-8111-111111111111",ExecutorID:"setjoin-updater",StateGeneration:9,State:state,Step:"fixture",Evidence:json.RawMessage(`{"fixture":"<Go>&","ordered":{"z":1,"a":2}}`)}); if err!=nil {panic(err)}
  // Keep the pre-compaction log too, exercising snapshot/log overlap verification.
  log,err:=os.ReadFile(filepath.Join(dir,"events.jsonl")); if err!=nil {panic(err)}
  var snapshot []byte
  if name != "active" {
   if err=journal.Compact(true);err!=nil {panic(err)}
   snapshot,err=os.ReadFile(filepath.Join(dir,"snapshot.json"));if err!=nil {panic(err)}
  }
  result[name]=map[string]string{"events":string(log),"snapshot":string(snapshot)}
  os.RemoveAll(dir)
 }
 path:=filepath.Join(os.Getenv("REL3_RECORD_DIR"),"legacy-journals.json")
 file,err:=os.OpenFile(path,os.O_CREATE|os.O_EXCL|os.O_WRONLY,0600);if err!=nil {panic(err)}
 if err=json.NewEncoder(file).Encode(map[string]interface{}{"go_source":os.Getenv("GOFREE_GO_SOURCE"),"cases":result});err!=nil {panic(err)}
 file.Close();fmt.Println("legacy journal corpus recorded");os.Exit(0)
}
